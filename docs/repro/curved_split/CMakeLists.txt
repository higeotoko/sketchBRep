cmake_minimum_required(VERSION 3.16)
project(curved_split_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE REQUIRED)
add_executable(curved_split_probe main.cpp)
target_include_directories(curved_split_probe SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(curved_split_probe PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(curved_split_probe PRIVATE TKBO TKBool TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
